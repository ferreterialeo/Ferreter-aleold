/* Configuración inicial de Supabase.
   Sustituye los dos valores de ejemplo por la URL y la clave PUBLICABLE/anon.
   Nunca uses service_role ni una clave secreta en este archivo. */
window.FERRETERIA_SUPABASE_URL = "PEGA_AQUI_LA_URL_DE_TU_PROYECTO";
window.FERRETERIA_SUPABASE_ANON_KEY = "PEGA_AQUI_TU_CLAVE_PUBLICA_ANON";

(function () {
  if (!window.supabase || !window.supabase.createClient) {
    console.error("No se cargó la librería de Supabase. Revisa el script CDN en index.html.");
    return;
  }
  if (window.FERRETERIA_SUPABASE_URL.startsWith("PEGA_AQUI") ||
      window.FERRETERIA_SUPABASE_ANON_KEY.startsWith("PEGA_AQUI")) {
    console.warn("Completa la URL y la clave pública en supabase-config.js.");
    return;
  }
  window.ferreteriaSupabase = window.supabase.createClient(
    window.FERRETERIA_SUPABASE_URL,
    window.FERRETERIA_SUPABASE_ANON_KEY
  );
})();
